The `analysis` values follow CycloneDX 1.6, the page body becomes the analysis `detail`. */}} {{- $page := . -}} {{- $path := $page.File.Path -}} {{- $ids := $page.Params.cve -}} {{- if not $ids -}} {{- errorf "%s: 'cve' must list at least one vulnerability ID" $path -}} {{- end -}} {{- if not (reflect.IsSlice $ids) -}} {{- $ids = slice $ids -}} {{- end -}} {{- $components := $page.Params.components -}} {{- if not $components -}} {{- errorf "%s: 'components' must list at least one package URL" $path -}} {{- end -}} {{- range $components -}} {{- if not (hasPrefix . "pkg:") -}} {{- errorf "%s: component %q is not a package URL" $path . -}} {{- end -}} {{- end -}} {{- $analysis := $page.Params.analysis -}} {{- $states := slice "resolved" "resolved_with_pedigree" "exploitable" "in_triage" "false_positive" "not_affected" -}} {{- $justifications := slice "code_not_present" "code_not_reachable" "requires_configuration" "requires_dependency" "requires_environment" "protected_by_compiler" "protected_at_runtime" "protected_at_perimeter" "protected_by_mitigating_control" -}} {{- $responses := slice "can_not_fix" "will_not_fix" "update" "rollback" "workaround_available" -}} {{- if not (in $states $analysis.state) -}} {{- errorf "%s: analysis.state %q must be one of %v" $path $analysis.state $states -}} {{- end -}} {{- with $analysis.justification -}} {{- if not (in $justifications .) -}} {{- errorf "%s: analysis.justification %q must be one of %v" $path . $justifications -}} {{- end -}} {{- end -}} {{- range $analysis.response -}} {{- if not (in $responses .) -}} {{- errorf "%s: analysis.response %q must be one of %v" $path . $responses -}} {{- end -}} {{- end -}} {{- $result := dict "state" $analysis.state "detail" (strings.TrimSpace $page.RawContent) -}} {{- with $analysis.justification -}} {{- $result = merge $result (dict "justification" .) -}} {{- end -}} {{- with $analysis.response -}} {{- $result = merge $result (dict "response" .) -}} {{- end -}} {{- return dict "ids" $ids "components" $components "versions" ($page.Params.versions | default "") "analysis" $result "date" $page.Date -}}
